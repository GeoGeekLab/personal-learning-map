function renderNav(){
 const u=UI[activeLang];
 const label=document.getElementById('index-label');
 if(label) label.textContent=activeLang==='zh'?'目录':'INDEX';
 const nav=document.getElementById('floating-nav');
 nav.innerHTML=u.nav.map(x=>`<button type="button" data-go="${x[2]}"><span class="n">${x[0]}</span><span class="t">${x[1]}</span></button>`).join('');
 nav.querySelectorAll('[data-go]').forEach(b=>b.addEventListener('click',()=>go(b.dataset.go)));
 updateActiveIndex();
}
function updateActiveIndex(){
 const page=document.querySelector(`.lang-page[data-lang="${activeLang}"]`);
 if(!page) return;
 const sections=[...page.querySelectorAll('[data-section]')].filter(el=>el.dataset.section!=='top');
 let current=sections[0]?.dataset.section;
 for(const el of sections){
   if(el.getBoundingClientRect().top<=150) current=el.dataset.section;
 }
 document.querySelectorAll('#floating-nav [data-go]').forEach(b=>b.classList.toggle('active',b.dataset.go===current));
}
window.addEventListener('scroll',()=>requestAnimationFrame(updateActiveIndex),{passive:true});
window.addEventListener('resize',()=>requestAnimationFrame(updateActiveIndex));

}
function setLang(lang){

 document.querySelectorAll('[data-lang-btn]').forEach(b=>b.classList.toggle('active',b.dataset.langBtn===lang));
 renderNav(); renderFilters(lang); renderResources(lang);
}
function go(section){
